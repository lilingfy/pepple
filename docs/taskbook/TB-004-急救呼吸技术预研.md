## 2. 需求背景

### 2.1 产品定位
急救呼吸是 Pebble AI 的情绪急救功能：
- 当用户情绪淹没时提供即时帮助
- 基于科学的 4-7-8 呼吸法
- 全屏沉浸式体验，屏蔽外界干扰
- 通过视觉律动引导用户找回平静

### 2.2 科学依据
4-7-8 呼吸法（Relaxing Breath）：
- 吸气 4 秒：激活副交感神经系统
- 屏息 7 秒：让氧气充分进入血液
- 呼气 8 秒：排出二氧化碳，放松身心

### 2.3 用户场景
1. 感到焦虑/愤怒/压力过大
2. 进入急救呼吸页面
3. 跟随鹅卵石律动呼吸
4. 建议持续 2 分钟（1:59 倒计时）

### 4.2 核心动画实现

#### 呼吸动画 Hook

```typescript
// hooks/useBreathing.ts
"use client";

import { useState, useCallback, useEffect } from 'react';
import { useAnimation } from 'framer-motion';

type BreathingPhase = 'inhale' | 'hold' | 'exhale' | 'idle';

interface BreathingState {
  phase: BreathingPhase;
  progress: number;
  isActive: boolean;
}

export function useBreathing() {
  const [phase, setPhase] = useState<BreathingPhase>('idle');
  const [isActive, setIsActive] = useState(false);
  const controls = useAnimation();

  // 4-7-8 呼吸周期配置
  const CYCLE = {
    inhale: 4000,   // 4秒
    hold: 7000,     // 7秒
    exhale: 8000,   // 8秒
  };

  const startBreathing = useCallback(async () => {
    if (isActive) return;
    setIsActive(true);

    const breatheCycle = async () => {
      // 吸气阶段 (4秒)
      setPhase('inhale');
      await controls.start({
        scale: 1.5,
        opacity: 1,
        transition: { duration: 4, ease: 'easeInOut' }
      });

      // 屏息阶段 (7秒)
      setPhase('hold');
      await new Promise(resolve => setTimeout(resolve, 7000));

      // 呼气阶段 (8秒)
      setPhase('exhale');
      await controls.start({

        scale: 1,
        opacity: 0.8,
        transition: { duration: 8, ease: 'easeInOut' }
      });

      // 循环
      if (isActive) {
        breatheCycle();
      }
    };

    breatheCycle();
  }, [controls, isActive]);

  const stopBreathing = useCallback(() => {
    setIsActive(false);
    setPhase('idle');
    controls.stop();
  }, [controls]);

  return {
    phase,
    isActive,
    controls,
    startBreathing,
    stopBreathing
  };
}
```

#### 倒计时 Hook

```typescript
// hooks/useCountdown.ts
"use client";

import { useState, useEffect, useCallback } from 'react';

interface CountdownState {
  minutes: number;
  seconds: number;
  isRunning: boolean;
  isComplete: boolean;
}

export function useCountdown(initialMinutes: number = 1, initialSeconds: number = 59) {
  const [state, setState] = useState<CountdownState>({
    minutes: initialMinutes,
    seconds: initialSeconds,
    isRunning: false,
    isComplete: false
  });

  const start = useCallback(() => {
    setState(prev => ({ ...prev, isRunning: true, isComplete: false }));
  }, []);

  const stop = useCallback(() => {
    setState(prev => ({ ...prev, isRunning: false }));
  }, []);

  const reset = useCallback(() => {
    setState({
      minutes: initialMinutes,
      seconds: initialSeconds,
      isRunning: false,
      isComplete: false
    });
  }, [initialMinutes, initialSeconds]);

  useEffect(() => {
    if (!state.isRunning || state.isComplete) return;

    const timer = setInterval(() => {
      setState(prev => {
        if (prev.seconds > 0) {
          return { ...prev, seconds: prev.seconds - 1 };
        }
        if (prev.minutes > 0) {
          return { ...prev, minutes: prev.minutes - 1, seconds: 59 };
        }
        return { ...prev, isRunning: false, isComplete: true };
      });
    }, 1000);

    return () => clearInterval(timer);
  }, [state.isRunning, state.isComplete]);

  const formattedTime = `${String(state.minutes).padStart(2, '0')}:${String(state.seconds).padStart(2, '0')}`;

  return {
    ...state,
    formattedTime,
    start,
    stop,
    reset
  };
}
```

"use client";

import { motion } from "framer-motion";

interface WaveLayersProps {
  isActive: boolean;
}

export function WaveLayers({ isActive }: WaveLayersProps) {
  if (!isActive) return null;

  return (
    <>
      {[0, 1.3, 2.6].map((delay, index) => (
        <motion.div
          key={index}
          className="absolute w-72 h-72 rounded-[45%_55%_60%_40%/50%_45%_55%_50%] bg-pebble-healing/20 pointer-events-none"
          initial={{ scale: 1, opacity: 0.3, filter: 'blur(8px)' }}
          animate={{
            scale: [1, 2.8],
            opacity: [0.3, 0],
            filter: ['blur(8px)', 'blur(30px)']
          }}
          transition={{
            duration: 4,
            repeat: Infinity,
            delay,
            ease: [0.2, 0.5, 0.3, 1]
          }}
        />
      ))}
    </>
  );
}
